		function populateAntiForgery(token) {

			var absPat = /^https?:\/\/|^\/\//i;
			var forms = document.getElementsByTagName("form");
			for (var i = 0; i < forms.length; i++) {
				if (forms[i].method == "post" &&
					!forms[i].querySelector('[name="__RequestVerificationToken"]') &&
					!absPat.test(forms[i].getAttribute("action"))) {
					forms[i].appendChild(input.cloneNode());
				}
			}
		}

			var xhr = new XMLHttpRequest();

				try {
					if (this.status !== 200) {
						throw Error("Invalid status. Could not retrieve antiforgery token.");

					callback(obj.token);

					typeof error === "function" && error(e);

        <h2 class="subhead1 ">Election Types</h2><p style="margin-left: 20px;">A<strong>&nbsp;general election&nbsp;</strong>is held on the 1st Tuesday after the first Monday in the month of November. The term &quot;general election&quot; is used principally to distinguish that election from &apos;primary elections&apos; or &apos;special elections.&apos; View the most <a aria-label="recent Election Results Opens in new window" class="standard_link" href="/571/Election-Results" rel="noopener noreferrer" target="_blank">recent Election Results</a>.</p><p style="margin-left: 20px;">A <strong>primary election&nbsp;</strong>is held in June&nbsp;prior to the general election. A primary election may take place within each of New York State&apos;s official political parties. It precedes the general election and provides enrolled political party members the opportunity to nominate their party&apos;s candidates for elected office as well as to elect various party officials. However, if there is no contest, there is no primary. Candidates of the same political party seek to appear on the ballot as the party&apos;s candidate for office.</p><p style="margin-left: 20px;">A <strong>special election</strong> is held in jurisdictions to fill an office that becomes vacant prior to the end of its corresponding term.</p><p><span class="subhead1">Working the Polls</span></p><p style="margin-left: 20px;">Want to work as an Election Day Inspector or Election Day Translator? The Board of Elections is accepting applications for Inspectors and Spanish Translators. Complete the application below and submit the form to the Board. You will be notified if your application has been accepted.</p><p style="margin-left: 20px;"><span class="style3"><a aria-label="Election Day Inspector (PDF) Opens in new window" class="standard_link" href="/DocumentCenter/View/2410" rel="noopener noreferrer" target="_blank" title="Election Day Inspector"><span class="standard_link">Election Day Inspector (PDF)</span></a>&nbsp;</span></p><p style="margin-left: 20px;"><a aria-label="Spanish Translators (PDF) Opens in new window" class="standard_link" href="/DocumentCenter/View/38576" rel="noopener" target="_blank" title="Election Day Inspector (Spanish)"><span class="standard_link">Spanish Translators (PDF)</span></a></p><p><span class="subhead1">Polling Locations&nbsp;</span></p><p><a name="Polling"></a>In April you will receive a notice from the Nassau County Board of Elections notifying you of your polling place, clearly marked on the face of the card. However:</p><ul><li class="content">If any of the information on this card is incorrect, please call the Board of Elections.</li><li class="content">If you have moved since your last registration, complete a new <a href="/DocumentCenter/View/40117" target="_blank" rel="noopener"></a><span class="standard_link"><a aria-label="registration form Opens in new window" href="/DocumentCenter/View/40117" rel="noopener" target="_blank">registration form</a></span><span class="standard_link">.&nbsp;</span></li><li class="content">If you have moved out of Nassau County, you must register in your new county.</li><li class="content">If you are physically disabled and your polling place is NOT accessible, you may request that your record be moved to an accessible location. The request must be made in writing and it must be sent to the Board prior to the 20th day before the election. Please note: All polling locations are equipped with Ballot Marking Devices for disabled voters.</li><li class="content">If you have not received a card or cannot locate it and want to verify your voting location, <a class="standard_link" href="https://voterlookup.elections.ny.gov/" rel="noopener noreferrer" target="_blank">visit the elections website</a><span class="style2">.</span></li></ul>
