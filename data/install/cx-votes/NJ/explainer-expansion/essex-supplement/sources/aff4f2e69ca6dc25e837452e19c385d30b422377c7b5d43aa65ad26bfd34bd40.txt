        function googleTranslateApplyCookie() {
            var match = document.cookie.match(/(?:^|;\s*)googtrans=([^;]+)/);
            if (!match) {
                return;
            }

            var parts = decodeURIComponent(match[1]).split('/').filter(function (segment) {
                return segment.length > 0;
            });

            if (parts.length < 2) {
                return;
            }

            var lang = parts[parts.length - 1];
            if (!lang || lang === 'en') {
                return;
            }

            var select = document.querySelector('.goog-te-combo');
            if (!select || !select.options || !select.options.length) {
                return;
            }

            var i;
            for (i = 0; i < select.options.length; i++) {
                if (select.options[i].value === lang) {
                    select.value = lang;
                    if (document.createEvent) {
                        var evt = document.createEvent('HTMLEvents');
                        evt.initEvent('change', true, true);
                        select.dispatchEvent(evt);
                    } else if (window.jQuery) {
                        window.jQuery(select).trigger('change');
                    }
                    break;
                }
            }
        }

        function googleTranslateElementInit() {
            new google.translate.TranslateElement({
                pageLanguage: 'en',
                includedLanguages: 'es,pt,zh-CN,zh-TW,ko,fr,it,pl,ar,hi,ru,de,vi,ht,tl,ur,bn',
                layout: google.translate.TranslateElement.InlineLayout.SIMPLE,
                autoDisplay: false
            }, 'google_translate_element');

            window.setTimeout(googleTranslateApplyCookie, 100);
            window.setTimeout(googleTranslateApplyCookie, 500);
            window.setTimeout(googleTranslateApplyCookie, 1500);

            if (window.jQuery) {
                jQuery(document).trigger('googleTranslateReady');
            }
        }
