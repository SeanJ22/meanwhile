import {readFileSync,writeFileSync,readdirSync,mkdirSync} from 'node:fs';
import {pathToFileURL} from 'node:url';
import {createHash} from 'node:crypto';
const root='C:/Users/seanj/dev/wtp-fleet-reactivation-4';
const mod=async path=>import(pathToFileURL(`${root}/${path}`));
const {createPgRepositories,controlPlaneUrlFromEnv}=await mod('lib/control-plane/repository/pg-store.ts');
const {installNeonCircuitBreaker}=await mod('lib/control-plane/neon-circuit.ts');
const {readResearchProgression}=await mod('lib/control-plane/research-progression.ts');
const {readActivationGate}=await mod('lib/control-plane/promotion/activation-gate.ts');
const {readCandidateStore,readMethodStore}=await mod('lib/method/candidate.ts');
const {readConclusionStore}=await mod('lib/method/conclusions.ts');
installNeonCircuitBreaker(root);
const repos=await createPgRepositories(controlPlaneUrlFromEnv(),new Date().toISOString(),{skipBootstrap:true});
const knowledge=readMethodStore(`${root}/data/method/knowledge.json`);
const candidates=readCandidateStore(`${root}/data/method/candidates.json`).candidates;
const conclusions=readConclusionStore(`${root}/data/method/conclusions.json`);
const gate=readActivationGate(root);
const outputDir=`${root}/data/ops/prompt4/replayed-windows`;
mkdirSync(outputDir,{recursive:true});
for(const name of readdirSync(`${root}/data/ops/prompt4/completed-windows`).filter(n=>n.endsWith('.json')).sort()) {
  const path=`${root}/data/ops/prompt4/completed-windows/${name}`;
  const bytes=readFileSync(path), source=JSON.parse(bytes);
  const previous=source.researchProgression;
  const projection=await readResearchProgression(repos,{
    windowId:previous.windowId,through:previous.through,knowledge,candidates,conclusions,gate,
    researchBoundary:{since:previous.since,previousCompletedWindowId:previous.previousCompletedWindowId},
  });
  const receipt={formatVersion:1,kind:'REPLAY_FROM_DURABLE_HISTORY_NOT_REPLACEMENT_OF_EXECUTED_RECEIPT',
    replayedAt:new Date().toISOString(),codeHead:'6ca06e13453917f02e518457cec8cf4291ccaf94',
    executedReceipt:`data/ops/prompt4/completed-windows/${name}`,
    executedReceiptSha256:createHash('sha256').update(bytes).digest('hex'),projection};
  writeFileSync(`${outputDir}/${name}`,JSON.stringify(receipt,null,2)+'\n');
  console.log(JSON.stringify({window:name,through:projection.through,activity:projection.activity,regressions:projection.regressions.length}));
}
