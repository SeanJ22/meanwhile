/**
 * Owner-approved residue repair R1–R3 (docs/evidence/runner-settled-route-residue-2026-09-22.md).
 *
 *   node --experimental-strip-types repair-residue.mts --plan    READ ONLY: preconditions + before-image
 *   node --experimental-strip-types repair-residue.mts --apply   under the nightly-window lease: one guarded
 *                                                                transaction, then the post-check + after-image
 *
 * Scope is exactly the rows named in the evidence file: 65 RouteTasks (R1), 8 Attempts (R2), one lease (R3).
 * R4 is NOT part of this repair: the deployed sweep must already have released all 65 leases (precondition).
 * Every guarded statement divides by zero unless it wrote exactly the approved number of rows, so any drift
 * aborts the whole transaction and nothing is written. Nothing is deleted; no disposition is written.
 */
import { createHash } from "node:crypto";
import { readFileSync, writeFileSync } from "node:fs";
import { createRequire } from "node:module";
import { join } from "node:path";

const WORKTREE = "C:/Users/seanj/dev/we-the-people/.claude/worktrees/happy-poincare-b03465";
const { neon } = createRequire(`${WORKTREE}/package.json`)("@neondatabase/serverless");
const { createPgRepositories } = await import(`file:///${WORKTREE}/lib/control-plane/repository/pg-store.ts`);
const { acquireWindowLease, releaseWindowLease, WINDOW_LEASE_ID } = await import(
  `file:///${WORKTREE}/lib/control-plane/promotion/window.ts`
);

const MODE = process.argv.includes("--apply") ? "apply" : "plan";
const OUT = process.argv.find((a) => a.startsWith("--out="))?.slice("--out=".length) ?? ".";
const STAMP = process.argv.find((a) => a.startsWith("--stamp="))?.slice("--stamp=".length) ?? "undated";
const env = Object.fromEntries(
  readFileSync("C:/Users/seanj/dev/we-the-people/.env.local", "utf8")
    .split(/\r?\n/)
    .filter((l) => /^[A-Z0-9_]+=/.test(l))
    .map((l) => {
      const i = l.indexOf("=");
      let v = l.slice(i + 1).trim();
      if ((v.startsWith('"') && v.endsWith('"')) || (v.startsWith("'") && v.endsWith("'"))) v = v.slice(1, -1);
      return [l.slice(0, i), v];
    }),
);
const URL_VALUE: string = env.WTP_CONTROL_PLANE_DATABASE_URL;
if (!URL_VALUE) throw new Error("WTP_CONTROL_PLANE_DATABASE_URL is not set in the main checkout .env.local");
const sql = neon(URL_VALUE);

const evidence = JSON.parse(
  readFileSync(join(WORKTREE, "docs/evidence/runner-settled-route-residue-2026-09-22.json"), "utf8"),
) as {
  claimedButDisposedRouteTasks: { routeTaskId: string; revision: number }[];
  neverClosedAttempts: { attemptId: string; revision: number }[];
  proofFixtureLease: { routeTaskId: string; revision: number; leaseId: string; holder: string };
};
const R1_IDS = evidence.claimedButDisposedRouteTasks.map((t) => t.routeTaskId).sort();
const R2_IDS = evidence.neverClosedAttempts.map((a) => a.attemptId).sort();
const PROOF = evidence.proofFixtureLease;
if (R1_IDS.length !== 65 || R2_IDS.length !== 8 || PROOF.routeTaskId !== "rt-a0c6bb05d955ba63cd7717ac") {
  throw new Error("the evidence file does not name exactly the approved 65 / 8 / 1 rows");
}
if (evidence.claimedButDisposedRouteTasks.some((t) => t.revision !== 4)) throw new Error("R1 evidence revision is not 4");
if (evidence.neverClosedAttempts.some((a) => a.revision !== 1)) throw new Error("R2 evidence revision is not 1");

/* ------------------------------------------------------------ preconditions (READ ONLY) */

const PRE = {
  r1: `SELECT count(*)::int AS n FROM cp_route_tasks rt
        WHERE rt.id = ANY($1::text[]) AND rt.revision = 4
          AND rt.value->>'state' = 'CLAIMED'
          AND jsonb_typeof(rt.value->'disposition') = 'object'
          AND NOT (rt.value ? 'workerTask')
          AND EXISTS (SELECT 1 FROM cp_leases l WHERE l.id = rt.id AND (l.value->>'released')::boolean = true)
          AND (SELECT count(*) FROM cp_log_dispositions d WHERE d.record->>'routeTaskId' = rt.id) = 1
          AND (SELECT d.record->'disposedTask'->>'state' FROM cp_log_dispositions d WHERE d.record->>'routeTaskId' = rt.id) = 'DISPOSED'
          AND (rt.value - 'state') = ((SELECT d.record->'disposedTask' FROM cp_log_dispositions d WHERE d.record->>'routeTaskId' = rt.id) - 'state')`,
  r1LeasesUnreleased: `SELECT count(*)::int AS n FROM cp_leases l WHERE l.id = ANY($1::text[]) AND (l.value->>'released')::boolean = false`,
  r2: `SELECT count(*)::int AS n FROM cp_attempts a JOIN cp_route_tasks rt ON rt.id = a.value->>'routeTaskId'
        WHERE a.id = ANY($1::text[]) AND a.revision = 1
          AND jsonb_typeof(a.value->'outcome') = 'null'
          AND jsonb_typeof(rt.value->'disposition') = 'object'
          AND jsonb_array_length(coalesce(a.value->'artifactIds', '[]'::jsonb)) = 0
          AND NOT EXISTS (SELECT 1 FROM cp_log_artifacts x WHERE x.record->>'attemptId' = a.id)
          AND NOT EXISTS (SELECT 1 FROM cp_log_costs c WHERE c.record->>'attemptId' = a.id)
          AND NOT EXISTS (SELECT 1 FROM cp_leases l WHERE l.id = rt.id AND l.value->>'id' = a.value->>'leaseId'
                            AND (l.value->>'released')::boolean = false AND (l.value->>'expiresAt')::timestamptz > now())`,
  r3: `SELECT count(*)::int AS n FROM cp_leases l
        WHERE l.id = $1 AND l.revision = $2 AND l.value->>'id' = $3 AND l.value->>'holder' = $4
          AND (l.value->>'released')::boolean = false
          AND NOT EXISTS (SELECT 1 FROM cp_attempts a WHERE a.value->>'routeTaskId' = l.id)`,
  otherClaimedDisposed: `SELECT rt.id FROM cp_route_tasks rt
        WHERE rt.value->>'state' = 'CLAIMED' AND jsonb_typeof(rt.value->'disposition') = 'object'
          AND NOT (rt.id = ANY($1::text[])) ORDER BY rt.id`,
  window: `SELECT value, revision FROM cp_leases WHERE id = 'nightly-window'`,
};

async function preconditions() {
  const [r1, r1Unreleased, r2, r3, others, window] = await sql.transaction(
    [
      sql.query(PRE.r1, [R1_IDS]),
      sql.query(PRE.r1LeasesUnreleased, [R1_IDS]),
      sql.query(PRE.r2, [R2_IDS]),
      sql.query(PRE.r3, [PROOF.routeTaskId, PROOF.revision, PROOF.leaseId, PROOF.holder]),
      sql.query(PRE.otherClaimedDisposed, [R1_IDS]),
      sql.query(PRE.window),
    ],
    { readOnly: true },
  );
  return {
    r1: r1[0].n as number,
    r1LeasesStillUnreleased: r1Unreleased[0].n as number,
    r2: r2[0].n as number,
    r3: r3[0].n as number,
    otherClaimedButDisposedOutOfScope: (others as { id: string }[]).map((o) => o.id),
    window: window[0] ?? null,
  };
}

/* ------------------------------------------------------------ images (READ ONLY) */

async function image(label: string) {
  const [tasks, leases, commits, attempts, proof] = await sql.transaction(
    [
      sql.query(`SELECT id, revision, updated_at, value FROM cp_route_tasks WHERE id = ANY($1::text[]) ORDER BY id`, [R1_IDS]),
      sql.query(`SELECT id, revision, updated_at, value FROM cp_leases WHERE id = ANY($1::text[]) ORDER BY id`, [R1_IDS]),
      sql.query(
        `SELECT log_key, record_id, position, recorded_at, record FROM cp_log_dispositions
          WHERE record->>'routeTaskId' = ANY($1::text[]) ORDER BY record->>'routeTaskId', position`,
        [R1_IDS],
      ),
      sql.query(`SELECT id, revision, updated_at, value FROM cp_attempts WHERE id = ANY($1::text[]) ORDER BY id`, [R2_IDS]),
      sql.query(`SELECT id, revision, updated_at, value FROM cp_leases WHERE id = $1`, [PROOF.routeTaskId]),
    ],
    { readOnly: true },
  );
  const body = { label, readAt: new Date().toISOString(), routeTasks: tasks, routeTaskLeases: leases, dispositionCommits: commits, attempts, proofLease: proof };
  const text = `${JSON.stringify(body, null, 1)}\n`;
  const sha256 = createHash("sha256").update(text).digest("hex");
  const path = join(OUT, `runner-settled-route-residue-${label}.json`);
  writeFileSync(path, text);
  return { path, sha256, body };
}

/* ------------------------------------------------------------ the guarded transaction */

const GUARD_WINDOW = `SELECT 1 / (CASE WHEN EXISTS (
    SELECT 1 FROM cp_leases WHERE id = 'nightly-window' AND value->>'holder' = $1
       AND (value->>'released')::boolean = false AND (value->>'expiresAt')::timestamptz > now())
  THEN 1 ELSE 0 END) AS window_held`;

const R1 = `WITH restored AS (
  UPDATE cp_route_tasks rt
     SET value = jsonb_set(rt.value, '{state}', '"DISPOSED"'), revision = rt.revision + 1, updated_at = $2::timestamptz
   WHERE rt.id = ANY($1::text[]) AND rt.revision = 4
     AND rt.value->>'state' = 'CLAIMED'
     AND jsonb_typeof(rt.value->'disposition') = 'object'
     AND NOT (rt.value ? 'workerTask')
     AND EXISTS (SELECT 1 FROM cp_leases l WHERE l.id = rt.id AND (l.value->>'released')::boolean = true)
     AND (SELECT count(*) FROM cp_log_dispositions d WHERE d.record->>'routeTaskId' = rt.id) = 1
     AND (SELECT d.record->'disposedTask'->>'state' FROM cp_log_dispositions d WHERE d.record->>'routeTaskId' = rt.id) = 'DISPOSED'
     AND (rt.value - 'state') = ((SELECT d.record->'disposedTask' FROM cp_log_dispositions d WHERE d.record->>'routeTaskId' = rt.id) - 'state')
  RETURNING rt.id)
SELECT 1 / (CASE WHEN count(*) = 65 THEN 1 ELSE 0 END) AS r1 FROM restored`;

const R2 = `WITH closed AS (
  UPDATE cp_attempts a
     SET value = a.value || jsonb_build_object(
           'outcome', 'LEASE_EXPIRED',
           'endedAt', $2::text,
           'abortReason',
             'Opened at ' || (a.value->>'startedAt') ||
             ' during 2026-09-01/02 and never closed. Its lease ' || (a.value->>'leaseId') ||
             ' stopped being renewed and a later claim replaced it. Its RouteTask had already been disposed ' ||
             (rt.value->'disposition'->>'disposition') || ' by Attempt ' || (rt.value->'disposition'->>'attemptId') ||
             ', so this Attempt reached no conclusion and none is recorded. No artifact or cost was attributed ' ||
             'to it. Closed from outside by the owner-approved residue repair of ' || $2::text || '.'),
         revision = a.revision + 1, updated_at = $2::timestamptz
    FROM cp_route_tasks rt
   WHERE a.id = ANY($1::text[]) AND a.revision = 1
     AND jsonb_typeof(a.value->'outcome') = 'null'
     AND rt.id = a.value->>'routeTaskId'
     AND jsonb_typeof(rt.value->'disposition') = 'object'
     AND NOT EXISTS (
       SELECT 1 FROM cp_leases l
        WHERE l.id = rt.id AND l.value->>'id' = a.value->>'leaseId'
          AND (l.value->>'released')::boolean = false
          AND (l.value->>'expiresAt')::timestamptz > $2::timestamptz)
  RETURNING a.id)
SELECT 1 / (CASE WHEN count(*) = 8 THEN 1 ELSE 0 END) AS r2 FROM closed`;

const R3 = `WITH released AS (
  UPDATE cp_leases l
     SET value = jsonb_set(l.value, '{released}', 'true'), revision = l.revision + 1, updated_at = $5::timestamptz
   WHERE l.id = $1 AND l.revision = $2
     AND l.value->>'id' = $3
     AND l.value->>'holder' = $4
     AND (l.value->>'released')::boolean = false
     AND NOT EXISTS (SELECT 1 FROM cp_attempts a WHERE a.value->>'routeTaskId' = l.id)
  RETURNING l.id)
SELECT 1 / (CASE WHEN count(*) = 1 THEN 1 ELSE 0 END) AS r3 FROM released`;

/* ------------------------------------------------------------ post-check (READ ONLY) */

const POST = {
  r1: `SELECT count(*)::int AS n FROM cp_route_tasks rt
        WHERE rt.id = ANY($1::text[]) AND rt.revision = 5 AND rt.value->>'state' = 'DISPOSED'
          AND rt.value = (SELECT d.record->'disposedTask' FROM cp_log_dispositions d WHERE d.record->>'routeTaskId' = rt.id)`,
  r2: `SELECT count(*)::int AS n FROM cp_attempts a
        WHERE a.id = ANY($1::text[]) AND a.revision = 2 AND a.value->>'outcome' = 'LEASE_EXPIRED'
          AND a.value->>'endedAt' = $2 AND length(a.value->>'abortReason') > 0`,
  r3: `SELECT count(*)::int AS n FROM cp_leases WHERE id = $1 AND revision = $2 AND (value->>'released')::boolean = true`,
  commits: `SELECT count(*)::int AS n FROM cp_log_dispositions WHERE record->>'routeTaskId' = ANY($1::text[])`,
};

async function postcheck(repairAt: string) {
  const [r1, r2, r3, commits] = await sql.transaction(
    [
      sql.query(POST.r1, [R1_IDS]),
      sql.query(POST.r2, [R2_IDS, repairAt]),
      sql.query(POST.r3, [PROOF.routeTaskId, PROOF.revision + 1]),
      sql.query(POST.commits, [R1_IDS]),
    ],
    { readOnly: true },
  );
  return { r1: r1[0].n, r2: r2[0].n, r3: r3[0].n, commitsOnR1Tasks: commits[0].n };
}

/* ------------------------------------------------------------ main */

const pre = await preconditions();
console.log(JSON.stringify({ mode: MODE, preconditions: pre }, null, 1));
const ready = pre.r1 === 65 && pre.r1LeasesStillUnreleased === 0 && pre.r2 === 8 && pre.r3 === 1;
if (!ready) {
  console.log("PRECONDITIONS DIFFER from the approved plan: nothing is written.");
  process.exit(2);
}
const before = await image(MODE === "apply" ? `before-${STAMP}` : `before-plan-${STAMP}`);
console.log(`before-image ${before.path} sha256 ${before.sha256}`);
if (MODE === "plan") process.exit(0);

const holder = `residue-repair@${process.pid}`;
const repos = await createPgRepositories(URL_VALUE, new Date().toISOString(), { skipBootstrap: true });
const repairAt = new Date().toISOString();
await acquireWindowLease(repos, holder, repairAt, 10 * 60_000);
console.log(`nightly-window lease acquired by ${holder} at ${repairAt} (${WINDOW_LEASE_ID})`);
try {
  const again = await preconditions();
  if (!(again.r1 === 65 && again.r1LeasesStillUnreleased === 0 && again.r2 === 8 && again.r3 === 1)) {
    console.log(JSON.stringify({ recheck: again }, null, 1));
    throw new Error("PRECONDITIONS CHANGED after the window lease was taken: nothing is written");
  }
  const result = await sql.transaction([
    sql.query(GUARD_WINDOW, [holder]),
    sql.query(R1, [R1_IDS, repairAt]),
    sql.query(R2, [R2_IDS, repairAt]),
    sql.query(R3, [PROOF.routeTaskId, PROOF.revision, PROOF.leaseId, PROOF.holder, repairAt]),
  ]);
  console.log(`transaction committed: ${JSON.stringify(result.map((r) => r[0]))}`);
  const post = await postcheck(repairAt);
  console.log(JSON.stringify({ postcheck: post }, null, 1));
  const after = await image(`after-${STAMP}`);
  console.log(`after-image ${after.path} sha256 ${after.sha256}`);
} finally {
  const released = await releaseWindowLease(repos, holder, new Date().toISOString());
  console.log(`nightly-window lease released: ${released}`);
}
