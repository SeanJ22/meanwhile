/**
 * R2 correction: the 8 Attempts R2 closed store `endedAt` as "2026-09-23 00:34:14.194+00" (Postgres's text
 * form of a timestamptz, because the one parameter was also cast ::timestamptz in the same statement) instead
 * of the ISO form endAttempt writes. This rewrites ONLY that representation — the same instant — in
 * `endedAt` and inside `abortReason`, on exactly those 8 rows, under the nightly-window lease.
 *
 *   node --experimental-strip-types repair-r2-iso.mts --plan | --apply  [--out=DIR] [--stamp=YYYY-MM-DD]
 */
import { createHash } from "node:crypto";
import { readFileSync, writeFileSync } from "node:fs";
import { createRequire } from "node:module";
import { join } from "node:path";

const WORKTREE = "C:/Users/seanj/dev/we-the-people/.claude/worktrees/happy-poincare-b03465";
const { neon } = createRequire(`${WORKTREE}/package.json`)("@neondatabase/serverless");
const { createPgRepositories } = await import(`file:///${WORKTREE}/lib/control-plane/repository/pg-store.ts`);
const { acquireWindowLease, releaseWindowLease } = await import(`file:///${WORKTREE}/lib/control-plane/promotion/window.ts`);

const MODE = process.argv.includes("--apply") ? "apply" : "plan";
const OUT = process.argv.find((a) => a.startsWith("--out="))?.slice("--out=".length) ?? ".";
const STAMP = process.argv.find((a) => a.startsWith("--stamp="))?.slice("--stamp=".length) ?? "undated";
const env = Object.fromEntries(
  readFileSync("C:/Users/seanj/dev/we-the-people/.env.local", "utf8")
    .split(/\r?\n/)
    .filter((l) => /^[A-Z0-9_]+=/.test(l))
    .map((l) => {
      const i = l.indexOf("=");
      let v = l.slice(i + 1).trim();
      if ((v.startsWith('"') && v.endsWith('"')) || (v.startsWith("'") && v.endsWith("'"))) v = v.slice(1, -1);
      return [l.slice(0, i), v];
    }),
);
const URL_VALUE: string = env.WTP_CONTROL_PLANE_DATABASE_URL;
const sql = neon(URL_VALUE);
const evidence = JSON.parse(readFileSync(join(WORKTREE, "docs/evidence/runner-settled-route-residue-2026-09-22.json"), "utf8"));
const IDS: string[] = evidence.neverClosedAttempts.map((a: { attemptId: string }) => a.attemptId).sort();
if (IDS.length !== 8) throw new Error("expected exactly the 8 R2 Attempts");

const STORED = "2026-09-23 00:34:14.194+00"; // what R2 wrote
const ISO = "2026-09-23T00:34:14.194Z"; // the same instant, as endAttempt writes it
if (new Date(Date.parse(STORED)).toISOString() !== ISO) throw new Error("STORED and ISO are not the same instant");

const PRE = `SELECT count(*)::int AS n FROM cp_attempts a
  WHERE a.id = ANY($1::text[]) AND a.revision = 2 AND a.value->>'outcome' = 'LEASE_EXPIRED'
    AND a.value->>'endedAt' = $2::text AND strpos(a.value->>'abortReason', $2::text) > 0`;
const GUARD_WINDOW = `SELECT 1 / (CASE WHEN EXISTS (
    SELECT 1 FROM cp_leases WHERE id = 'nightly-window' AND value->>'holder' = $1::text
       AND (value->>'released')::boolean = false AND (value->>'expiresAt')::timestamptz > now())
  THEN 1 ELSE 0 END) AS window_held`;
const FIX = `WITH fixed AS (
  UPDATE cp_attempts a
     SET value = a.value || jsonb_build_object(
           'endedAt', $3::text,
           'abortReason', replace(a.value->>'abortReason', $2::text, $3::text)),
         revision = a.revision + 1, updated_at = now()
   WHERE a.id = ANY($1::text[]) AND a.revision = 2 AND a.value->>'outcome' = 'LEASE_EXPIRED'
     AND a.value->>'endedAt' = $2::text AND strpos(a.value->>'abortReason', $2::text) > 0
  RETURNING a.id)
SELECT 1 / (CASE WHEN count(*) = 8 THEN 1 ELSE 0 END) AS r2_iso FROM fixed`;
const POST = `SELECT count(*)::int AS n FROM cp_attempts a
  WHERE a.id = ANY($1::text[]) AND a.revision = 3 AND a.value->>'outcome' = 'LEASE_EXPIRED'
    AND a.value->>'endedAt' = $2::text
    AND a.value->>'endedAt' ~ '^\\d{4}-\\d\\d-\\d\\dT\\d\\d:\\d\\d:\\d\\d\\.\\d{3}Z$'
    AND strpos(a.value->>'abortReason', $2::text) > 0
    AND strpos(a.value->>'abortReason', $3::text) = 0
    AND strpos(a.value->>'abortReason', '+00') = 0`;
/* Class-wide: the same formatting defect anywhere in the timestamp fields R1–R3 touched or neighbour. */
const ISO_Z = `'^\\d{4}-\\d\\d-\\d\\dT\\d\\d:\\d\\d:\\d\\d(\\.\\d{1,6})?Z$'`;
const CLASS = {
  attemptsEndedAt: `SELECT count(*)::int AS n FROM cp_attempts WHERE jsonb_typeof(value->'endedAt') = 'string' AND NOT (value->>'endedAt' ~ ${ISO_Z})`,
  attemptsStartedAt: `SELECT count(*)::int AS n FROM cp_attempts WHERE jsonb_typeof(value->'startedAt') = 'string' AND NOT (value->>'startedAt' ~ ${ISO_Z})`,
  routeTaskDispositionAt: `SELECT count(*)::int AS n FROM cp_route_tasks WHERE jsonb_typeof(value->'disposition'->'at') = 'string' AND NOT (value->'disposition'->>'at' ~ ${ISO_Z})`,
  leasesExpiresAt: `SELECT count(*)::int AS n FROM cp_leases WHERE jsonb_typeof(value->'expiresAt') = 'string' AND NOT (value->>'expiresAt' ~ ${ISO_Z})`,
  leasesClaimedAt: `SELECT count(*)::int AS n FROM cp_leases WHERE jsonb_typeof(value->'claimedAt') = 'string' AND NOT (value->>'claimedAt' ~ ${ISO_Z})`,
  leasesHeartbeatAt: `SELECT count(*)::int AS n FROM cp_leases WHERE jsonb_typeof(value->'heartbeatAt') = 'string' AND NOT (value->>'heartbeatAt' ~ ${ISO_Z})`,
};

async function image(label: string) {
  const [rows] = await sql.transaction(
    [sql.query(`SELECT id, revision, updated_at, value FROM cp_attempts WHERE id = ANY($1::text[]) ORDER BY id`, [IDS])],
    { readOnly: true },
  );
  const text = `${JSON.stringify({ label, readAt: new Date().toISOString(), attempts: rows }, null, 1)}\n`;
  const path = join(OUT, `runner-settled-route-residue-${label}.json`);
  writeFileSync(path, text);
  return { path, sha256: createHash("sha256").update(text).digest("hex") };
}

const [pre] = await sql.transaction([sql.query(PRE, [IDS, STORED])], { readOnly: true });
console.log(JSON.stringify({ mode: MODE, preconditionRows: pre[0].n }));
if (pre[0].n !== 8) { console.log("PRECONDITIONS DIFFER: nothing is written."); process.exit(2); }
if (MODE === "plan") process.exit(0);

const holder = `residue-repair-r2-iso@${process.pid}`;
const repos = await createPgRepositories(URL_VALUE, new Date().toISOString(), { skipBootstrap: true });
await acquireWindowLease(repos, holder, new Date().toISOString(), 10 * 60_000);
console.log(`nightly-window lease acquired by ${holder}`);
try {
  const [again] = await sql.transaction([sql.query(PRE, [IDS, STORED])], { readOnly: true });
  if (again[0].n !== 8) throw new Error("PRECONDITIONS CHANGED under the lease: nothing is written");
  const result = await sql.transaction([sql.query(GUARD_WINDOW, [holder]), sql.query(FIX, [IDS, STORED, ISO])]);
  console.log(`transaction committed: ${JSON.stringify(result.map((r) => r[0]))}`);
  const [post] = await sql.transaction([sql.query(POST, [IDS, ISO, STORED])], { readOnly: true });
  console.log(JSON.stringify({ postcheckRows: post[0].n }));
  const names = Object.keys(CLASS) as (keyof typeof CLASS)[];
  const counts = await sql.transaction(names.map((k) => sql.query(CLASS[k])), { readOnly: true });
  console.log(JSON.stringify({ classWideNonIsoZ: Object.fromEntries(names.map((k, i) => [k, counts[i][0].n])) }));
  const after = await image(`after-iso-${STAMP}`);
  console.log(`after-image ${after.path} sha256 ${after.sha256}`);
} finally {
  console.log(`nightly-window lease released: ${await releaseWindowLease(repos, holder, new Date().toISOString())}`);
}
