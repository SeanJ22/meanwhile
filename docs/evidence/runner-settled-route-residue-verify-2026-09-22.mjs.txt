// READ-ONLY post-deploy verification of the settled-route prevention and the 62 recoveries.
//   node verify-recovery.mjs <sinceIso> [out.json]
// <sinceIso>: the moment the prevention started running (the first cycle on the deployed SHA).
// One READ ONLY transaction; the connection string is read by NAME and never printed.
import { readFileSync, writeFileSync } from "node:fs";
import { createRequire } from "node:module";

const WORKTREE = "C:/Users/seanj/dev/we-the-people/.claude/worktrees/happy-poincare-b03465";
const { neon } = createRequire(`${WORKTREE}/package.json`)("@neondatabase/serverless");
const env = Object.fromEntries(
  readFileSync("C:/Users/seanj/dev/we-the-people/.env.local", "utf8")
    .split(/\r?\n/)
    .filter((l) => /^[A-Z0-9_]+=/.test(l))
    .map((l) => {
      const i = l.indexOf("=");
      let v = l.slice(i + 1).trim();
      if ((v.startsWith('"') && v.endsWith('"')) || (v.startsWith("'") && v.endsWith("'"))) v = v.slice(1, -1);
      return [l.slice(0, i), v];
    }),
);
const sql = neon(env.WTP_CONTROL_PLANE_DATABASE_URL);
const since = process.argv[2];
if (!since || !Number.isFinite(Date.parse(since))) throw new Error("usage: verify-recovery.mjs <sinceIso> [out.json]");
const evidence = JSON.parse(readFileSync(`${WORKTREE}/docs/evidence/runner-settled-route-residue-2026-09-22.json`, "utf8"));
const OPEN62 = evidence.openOccurrencesOnDisposedRouteTasks;
const OCC = OPEN62.map((o) => o.occurrenceId);
const RT62 = OPEN62.map((o) => o.routeTaskId);
const OB62 = OPEN62.map((o) => o.obligationId);
const DISPOSED = `jsonb_typeof(rt.value->'disposition') = 'object'`;

const queries = {
  readAt: `SELECT now() AS read_at`,
  // 1. No settled task is claimed or walked again: Attempts started since the deploy on a task whose
  //    disposition was committed BEFORE that Attempt started (the disposing Attempt itself excluded).
  attemptsOnSettledSince: `SELECT a.id, a.value->>'routeTaskId' AS route_task, a.value->>'startedAt' AS started_at, a.value->>'outcome' AS outcome
      FROM cp_attempts a JOIN cp_route_tasks rt ON rt.id = a.value->>'routeTaskId'
     WHERE (a.value->>'startedAt')::timestamptz >= $1::timestamptz AND ${DISPOSED}
       AND a.id <> rt.value->'disposition'->>'attemptId'
       AND (rt.value->'disposition'->>'at')::timestamptz < (a.value->>'startedAt')::timestamptz`,
  claimedButDisposed: `SELECT rt.id FROM cp_route_tasks rt WHERE ${DISPOSED} AND rt.value->>'state' <> 'DISPOSED' ORDER BY rt.id`,
  // 2. The 62: occurrence closure, attempts and lease per route task.
  occurrences62: `SELECT o.id, o.value->>'obligationId' AS obligation, o.value->>'closedAt' AS closed_at,
        o.value->>'disposition' AS occ_disposition, left(coalesce(o.value->>'unresolvedReason', ''), 120) AS reason,
        rt.value->>'state' AS task_state, rt.revision::int AS task_revision, rt.value->'disposition'->>'disposition' AS task_disposition,
        (SELECT count(*)::int FROM cp_attempts a WHERE a.value->>'routeTaskId' = rt.id) AS attempts,
        l.revision::int AS lease_revision, (l.value->>'released')::boolean AS lease_released
      FROM cp_occurrences o JOIN cp_route_tasks rt ON rt.id = o.value->>'routeTaskId'
      LEFT JOIN cp_leases l ON l.id = rt.id
     WHERE o.id = ANY($1::text[]) ORDER BY o.id`,
  // 4. Unfinished obligations stay eligible: the obligation rows and their debt.
  obligations62: `SELECT ob.id, (ob.value->>'active')::boolean AS active, ob.value->>'openOccurrenceId' AS open_occurrence,
        ob.value->>'lastWorkedAt' AS last_worked, ob.value->>'nextDueAt' AS next_due, (ob.value->>'occurrenceCount')::int AS occurrences,
        (d.value->>'consecutiveFailures')::int AS debt_failures, d.value->>'retryAt' AS retry_at, left(coalesce(d.value->>'detail', ''), 100) AS debt_detail
      FROM cp_obligations ob LEFT JOIN cp_debts d ON d.id = ob.id
     WHERE ob.id = ANY($1::text[]) ORDER BY ob.id`,
  commits62: `SELECT d.record->>'routeTaskId' AS route_task, count(*)::int AS commits
      FROM cp_log_dispositions d WHERE d.record->>'routeTaskId' = ANY($1::text[]) GROUP BY 1`,
  cycles: `SELECT recorded_at, record->'runtimeOutcome'->>'startedAt' AS started_at, record->'runtimeOutcome'->>'state' AS state,
        record->'runtimeOutcome'->>'testedSha' AS sha,
        (SELECT string_agg(l, ' | ') FROM jsonb_array_elements_text(record->'lines') l
          WHERE l LIKE '  recovered %' OR l LIKE '  leases %' OR l LIKE '  dispatched %' OR l LIKE '  dispositions %') AS lines
      FROM cp_log_owner_digests
     WHERE record->'runtimeOutcome'->>'kind' = 'FRESHNESS_CYCLE' AND recorded_at >= $1::timestamptz
     ORDER BY recorded_at`,
};
const names = Object.keys(queries);
const params = {
  readAt: [], attemptsOnSettledSince: [since], claimedButDisposed: [], occurrences62: [OCC],
  obligations62: [OB62], commits62: [RT62], cycles: [since],
};
const results = await sql.transaction(names.map((k) => sql.query(queries[k], params[k])), { readOnly: true });
const out = {};
names.forEach((k, i) => (out[k] = results[i]));
if (process.argv[3]) writeFileSync(process.argv[3], `${JSON.stringify(out, null, 1)}\n`);

const g = (rows, f) => rows.reduce((m, r) => ((m[f(r)] = (m[f(r)] ?? 0) + 1), m), {});
const occ = out.occurrences62;
console.log(`read at ${new Date(out.readAt[0].read_at).toISOString()}, since ${since}`);
console.log(`1. Attempts started since on already-settled tasks: ${out.attemptsOnSettledSince.length}`);
console.log(`   CLAIMED-but-disposed tasks now: ${out.claimedButDisposed.length}`);
console.log(`2. The 62: closed ${occ.filter((o) => o.closed_at).length}, open ${occ.filter((o) => !o.closed_at).length}`);
console.log(`   closure: ${JSON.stringify(g(occ, (o) => `${o.task_disposition}->${o.closed_at ? (o.occ_disposition ?? "UNRESOLVED") : "OPEN"}`))}`);
console.log(`   attempts per task: ${JSON.stringify(g(occ, (o) => o.attempts))}; task state: ${JSON.stringify(g(occ, (o) => o.task_state))}; lease released: ${JSON.stringify(g(occ, (o) => o.lease_released))}`);
console.log(`   commits per task: ${JSON.stringify(g(out.commits62, (c) => c.commits))}`);
const ob = out.obligations62;
console.log(`4. obligations active: ${JSON.stringify(g(ob, (o) => o.active))}; lastWorkedAt: ${JSON.stringify(g(ob, (o) => o.last_worked))}; debt failures: ${JSON.stringify(g(ob, (o) => o.debt_failures))}`);
console.log(`   open occurrence now: ${JSON.stringify(g(ob, (o) => (o.open_occurrence === null ? "none" : OCC.includes(o.open_occurrence) ? "the settled one" : "a NEW one")))}`);
for (const c of out.cycles) console.log(`cycle ${c.started_at} ${c.state} sha ${c.sha?.slice(0, 8)}: ${c.lines}`);
